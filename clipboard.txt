import subprocess, time, os
ENV = dict(os.environ)
ENV['DISPLAY'] = ':0'

test_text = "测试ABC123"

# 先把文字放入剪贴板
def set_clip(text):
    p = subprocess.Popen(['xclip', '-selection', 'clipboard'], stdin=subprocess.PIPE, env=ENV)
    p.communicate(input=text.encode('utf-8'))
    time.sleep(0.3)

methods = [
    ("1. xdotool type delay=100", lambda: subprocess.run(['xdotool', 'search', '--class', 'chrome', 'type', '--clearmodifiers', '--delay', '100', test_text], env=ENV)),
    ("2. xdotool type delay=200", lambda: subprocess.run(['xdotool', 'search', '--class', 'chrome', 'type', '--clearmodifiers', '--delay', '200', test_text], env=ENV)),
    ("3. ctrl+v", lambda: [set_clip(test_text), subprocess.run(['xdotool', 'search', '--class', 'chrome', 'key', '--clearmodifiers', 'ctrl+v'], env=ENV)]),
    ("4. shift+Insert", lambda: [set_clip(test_text), subprocess.run(['xdotool', 'search', '--class', 'chrome', 'key', '--clearmodifiers', 'shift+Insert'], env=ENV)]),
    ("5. xdg-open 粘贴URL方式", lambda: [set_clip(test_text), subprocess.run(['xdotool', 'search', '--class', 'chrome', 'key', '--clearmodifiers', 'ctrl+shift+v'], env=ENV)]),
]

print('5秒后开始测试，请点击Claude.ai输入框！')
time.sleep(5)

for name, fn in methods:
    print(f'\n>>> 测试: {name}')
    fn()
    print('等5秒...')
    time.sleep(5)

print('测试完成！告诉我哪个方法有效 ✅')
