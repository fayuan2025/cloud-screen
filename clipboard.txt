import subprocess, time, os

ENV = dict(os.environ)
ENV['DISPLAY'] = ':0'

# 先把测试文字放入剪贴板
p = subprocess.Popen(['xclip', '-selection', 'clipboard'], stdin=subprocess.PIPE, env=ENV)
p.communicate(input='测试粘贴123'.encode('utf-8'))
time.sleep(0.5)

methods = [
    ("1. search --name 云拷贝 + ctrl+v",
     ['xdotool', 'search', '--name', '云拷贝', 'key', '--clearmodifiers', 'ctrl+v']),
    ("2. search --name Google + ctrl+v",
     ['xdotool', 'search', '--name', 'Google', 'key', '--clearmodifiers', 'ctrl+v']),
    ("3. search --class chrome + key ctrl+v",
     ['xdotool', 'search', '--class', 'chrome', 'key', '--clearmodifiers', 'ctrl+v']),
    ("4. type 直接输入",
     ['xdotool', 'search', '--name', '云拷贝', 'type', '--clearmodifiers', '测试输入456']),
    ("5. focus + ctrl+v (77594651)",
     ['xdotool', 'windowfocus', '77594651']),
    ("6. focus + ctrl+v (79691777)",
     ['xdotool', 'windowfocus', '79691777']),
    ("7. focus + ctrl+v (79691779)",
     ['xdotool', 'windowfocus', '79691779']),
]

for name, cmd in methods[:4]:
    print(f'试: {name}')
    r = subprocess.run(cmd, env=ENV, capture_output=True, text=True)
    print(f'  stdout: {r.stdout.strip()}')
    print(f'  stderr: {r.stderr.strip()}')
    time.sleep(1)

# 方法5-7: windowfocus 后发 ctrl+v
for name, cmd in methods[4:]:
    print(f'试: {name}')
    r = subprocess.run(cmd, env=ENV, capture_output=True, text=True)
    print(f'  stderr: {r.stderr.strip()}')
    time.sleep(0.3)
    r2 = subprocess.run(['xdotool', 'key', '--clearmodifiers', 'ctrl+v'], env=ENV, capture_output=True, text=True)
    print(f'  ctrl+v stderr: {r2.stderr.strip()}')
    time.sleep(1)

print('测试完成，看 Doc 里哪个方法有效 ✅')
