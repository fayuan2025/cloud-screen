问题是 `google-chrome --new-tab` 会开新的 Chrome 进程，加上原来的就变成两个。

改用 `ctrl+t` + 输入地址，但这次先 `xdotool getactivewindow` 确保焦点正确：

```bash
python3 << 'PYEOF'
with open('/home/user/cloud-screen/main.py', 'r') as f:
    content = f.read()

old = """def paste_to_doc():
    doc_url = 'https://docs.google.com/document/d/1cV6eqGBgVLj-xK5Z1opY9mr4Sm77zOfjtITPjSlXq1A/edit'
    subprocess.Popen(['google-chrome', '--new-tab', doc_url], env=ENV)
    time.sleep(5)
    subprocess.run(['xdotool', 'search', '--class', 'chrome', 'key', '--clearmodifiers', 'ctrl+v'], env=ENV)
    print('已粘贴到Doc')"""

new = """def paste_to_doc():
    doc_url = 'https://docs.google.com/document/d/1cV6eqGBgVLj-xK5Z1opY9mr4Sm77zOfjtITPjSlXq1A/edit'
    # 激活Chrome，开新标签，输入地址
    wid = subprocess.run(['xdotool', 'search', '--class', 'chrome'], capture_output=True, text=True, env=ENV).stdout.strip().split()[-1]
    subprocess.run(['xdotool', 'windowfocus', wid], env=ENV)
    time.sleep(0.3)
    subprocess.run(['xdotool', 'key', 'ctrl+t'], env=ENV)
    time.sleep(0.5)
    subprocess.run(['xdotool', 'type', '--clearmodifiers', doc_url], env=ENV)
    time.sleep(0.3)
    subprocess.run(['xdotool', 'key', 'Return'], env=ENV)
    time.sleep(5)
    subprocess.run(['xdotool', 'key', 'ctrl+v'], env=ENV)
    print('已粘贴到Doc')"""

content = content.replace(old, new)
with open('/home/user/cloud-screen/main.py', 'w') as f:
    f.write(content)
print('完成 ✅')
PYEOF
```

重启：
```bash
pkill -f main.py && sleep 1 && cd ~/cloud-screen && source venv/bin/activate && python3 main.py &
```
